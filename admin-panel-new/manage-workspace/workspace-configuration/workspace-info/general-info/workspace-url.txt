1.
Title: Should not set workspace url by Workspace Admin

Steps:
# Login to the workspace as Workspace Admin
# Go to Manage Workspace > Workspace Configuration > Workspace Info page

Expectation:
Workspace URL field will remain disabled

2.
Title: Should set workspace url by Superuser via sudo

Steps:
# Login to the workspace as Superuser via sudo
# Go to Manage Workspace > Workspace Configuration > Workspace Info page
# Input workspace url in the Workspace URL field
# Click on the Save button

Expectation:
The success alert will be shown
Workspace URL will be updated

3.

# Login to the workspace as Superuser via sudo
# Go to Manage Workspace > Workspace Configuration > Workspace Info page
# Input blank space in "Workspace URL" field
# Click on the Save button

Expectation:
Error message will be shown

Steps:
# Input html tag in "Workspace URL" field
# Click on the Save button

Expectation:
Error message will be shown

Steps:
# Input special characters in "Workspace URL" field
# Click on the Save button

Error message will be shown

Steps:

4.

Steps:
# Login to the workspace as Superuser via sudo
# Go to Manage Workspace > Workspace Configuration > Workspace Info page
